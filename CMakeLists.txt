cmake_minimum_required(VERSION 3.20)
project(EmeraldGlassPreview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Widgets OpenGLWidgets OpenGL)
qt_add_executable(emerald-glass-preview preview/main.cpp)
target_link_libraries(emerald-glass-preview PRIVATE Qt6::Widgets Qt6::OpenGLWidgets Qt6::OpenGL)
target_compile_definitions(emerald-glass-preview PRIVATE PREVIEW_SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/preview")

option(BUILD_KWIN_DECORATION "Build the native Emerald Glass decoration" ON)
if(BUILD_KWIN_DECORATION)
    include(CTest)
    add_subdirectory(native)
    target_sources(emerald-glass-preview PRIVATE preview/sheetcomparison.cpp)
    find_package(KF6CoreAddons REQUIRED)
    find_package(KF6KCMUtils REQUIRED)
    target_link_libraries(emerald-glass-preview PRIVATE emeraldglass-material-preview KF6::CoreAddons KF6::KCMUtils)
    add_dependencies(emerald-glass-preview kcm_emeraldglassdecoration)
    target_compile_definitions(emerald-glass-preview PRIVATE KCM_PLUGIN_PATH="$<TARGET_FILE:kcm_emeraldglassdecoration>")
    target_compile_definitions(emerald-glass-preview PRIVATE HAS_MATERIAL_PREVIEW)
endif()
