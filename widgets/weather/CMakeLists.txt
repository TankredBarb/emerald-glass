cmake_minimum_required(VERSION 3.20)
project(EmeraldWeather LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Quick Network ShaderTools)
set(package_dir "${CMAKE_CURRENT_BINARY_DIR}/package")
configure_file(package/metadata.json "${package_dir}/metadata.json" COPYONLY)
configure_file(README.md "${package_dir}/README.md" COPYONLY)
configure_file(../../LICENSE "${package_dir}/LICENSE" COPYONLY)
configure_file(../../COPYING "${package_dir}/COPYING" COPYONLY)
foreach(component main WeatherCard ForecastPopup ConfigLocation ConfigText)
    configure_file("package/contents/ui/${component}.qml" "${package_dir}/contents/ui/${component}.qml" COPYONLY)
endforeach()
# Both independent plasmoids ship the same glass and editor implementation.
foreach(component GlassCard DesktopRotationFix ConfigEffects EffectSlider TextFontButton SensorText)
    configure_file("../systemmonitor/package/contents/ui/${component}.qml" "${package_dir}/contents/ui/${component}.qml" COPYONLY)
endforeach()
configure_file(Preview.qml "${package_dir}/contents/ui/Preview.qml" COPYONLY)
foreach(config main.xml config.qml)
    configure_file("package/contents/config/${config}" "${package_dir}/contents/config/${config}" COPYONLY)
endforeach()
configure_file(backend.qmldir "${package_dir}/contents/ui/backend/qmldir" COPYONLY)
add_custom_command(OUTPUT "${package_dir}/contents/ui/glass.frag.qsb"
    COMMAND Qt6::qsb --qt6 -o "${package_dir}/contents/ui/glass.frag.qsb" "${CMAKE_CURRENT_SOURCE_DIR}/../systemmonitor/glass.frag"
    DEPENDS ../systemmonitor/glass.frag VERBATIM)
add_custom_target(weather-glass ALL DEPENDS "${package_dir}/contents/ui/glass.frag.qsb")
add_library(emeraldweatherbackend MODULE weatherbackend.cpp weatherbackend.h weatherdata.cpp weatherdata.h plugin.cpp)
set_target_properties(emeraldweatherbackend PROPERTIES AUTOMOC ON
    LIBRARY_OUTPUT_DIRECTORY "${package_dir}/contents/ui/backend")
target_link_libraries(emeraldweatherbackend PRIVATE Qt6::Quick Qt6::Network)
add_executable(emerald-weather-preview preview.cpp)
target_link_libraries(emerald-weather-preview PRIVATE Qt6::Quick)
target_compile_definitions(emerald-weather-preview PRIVATE
    WEATHER_UI="${package_dir}/contents/ui")
add_dependencies(emerald-weather-preview weather-glass emeraldweatherbackend)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test QuickTest)
    add_executable(weather-data-test tests/weather_data_test.cpp weatherdata.cpp weatherbackend.cpp weatherbackend.h)
    set_target_properties(weather-data-test PROPERTIES AUTOMOC ON)
    target_include_directories(weather-data-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(weather-data-test PRIVATE Qt6::Test Qt6::Network)
    add_test(NAME weather-data COMMAND weather-data-test)
    add_executable(weather-qml-test tests/qml_main.cpp)
    target_link_libraries(weather-qml-test PRIVATE Qt6::QuickTest)
    configure_file(tests/tst_weather.qml tests/tst_weather.qml COPYONLY)
    target_compile_definitions(weather-qml-test PRIVATE QUICK_TEST_SOURCE_DIR="${CMAKE_CURRENT_BINARY_DIR}/tests")
    add_dependencies(weather-qml-test emeraldweatherbackend weather-glass)
    add_test(NAME weather-ui COMMAND weather-qml-test)
    set_tests_properties(weather-data weather-ui PROPERTIES TIMEOUT 30)
    set_tests_properties(weather-ui PROPERTIES ENVIRONMENT "QT_QUICK_CONTROLS_STYLE=org.kde.desktop")
endif()
